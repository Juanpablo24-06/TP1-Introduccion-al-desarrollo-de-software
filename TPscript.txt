#!/bin/bash

if [ "$1" = "-d" ] 
   then
   pkill -f consolidar.sh
   rm -r "EPNr1"
        exit
fi

if [ -n "$FILENAME" ]
  then
mensaje=""
while [ "$opcion" != "7" ]
do
   echo " "
   echo "seleccione una opción:"
   echo " "
   echo "1) crear entorno"
   echo "2) correr proceso"
   echo "3) Listado de alumnos ordenados por número de padrón"
   echo "4) Las 10 notas mas altas"
   echo "5) Buscar padron"
   echo "6) Mostrar log"
   echo "7) Salir"
   echo " "
   echo "$mensaje"
   echo " "
   echo -n "opcion elegida: "
   read opcion


   case "$opcion" in
1)
        mkdir -p EPNr1
        mkdir -p EPNr1/Entrada
        mkdir -p EPNr1/Salida
        mkdir -p EPNr1/procesado
        touch "EPNr1/Salida/$FILENAME.txt"
        mensaje="Entorno generado correctamente"
clear
        ;;
        
2)
if pgrep -f consolidar.sh > /dev/null
  then
    clear
    mensaje="ya hay un proceso corriendo en este momento, si quieres cerrarlos ejecuta la opcion 7"
  else
        if [ -d "EPNr1" ]
        then
             touch EPNr1/consolidar.sh
             cat > EPNr1/consolidar.sh << finaliza

#!/bin/bash
while [ -d "EPNr1" ]
do
for archivo in EPNr1/Entrada/*.txt
do
  if [ -f "\$archivo" ]
  then
      cat "\$archivo" >> EPNr1/Salida/$FILENAME.txt
      mv "\$archivo" EPNr1/procesado
      echo "\$(date '+%d/%m/%Y %H:%M:%S') - Procesado archivo \$(basename "\$archivo")" >> EPNr1/procesado.log 
  fi
done

sleep 5
done
finaliza
             bash EPNr1/consolidar.sh &
             PID=$! 
             mensaje="El proceso esta corriendo"
        clear
        else
             clear
             mensaje="Primero cree el entorno"
        fi
fi
        ;;
        
3)
             if [ -f "EPNr1/Salida/$FILENAME.txt" ]
             then
             sort -n -t' ' -k1 EPNr1/Salida/$FILENAME.txt
             else
               if [ -d "EPNr1" ]
               then 
                 mensaje="No se encuentra el archivo necesario en la carpeta Salida" 
               else
                 mensaje="No existe la carpeta EPNr1, generala con la opción 1"
               fi
             fi
        ;;
4)
             if [ -f EPNr1/Salida/$FILENAME.txt ]
             then
             sort -nr -t' ' -k5 EPNr1/Salida/$FILENAME.txt | head
             else
               if [ -d "EPNr1" ]
               then 
                 mensaje="No se encuentra el archivo necesario en la carpeta Salida" 
               else
                 mensaje="No existe la carpeta EPNr1, generala con la opción 1"
               fi
             fi
        ;;
5)
if [ -f EPNr1/Salida/$FILENAME.txt ]
  then
        clear
        echo -n "Ingrese el padron que esta buscando: "
        read padron
        grep  "^$padron " "EPNr1/Salida/$FILENAME.txt"
  else
    if [ -d "EPNr1" ]
             then 
                 mensaje="No se encuentra el archivo necesario en la carpeta Salida" 
             else
                 mensaje="No existe la carpeta EPNr1, generala con la opción 1"
    fi
fi
        ;;
6)
if [ -f EPNr1/procesado.log ]
  then 
        clear
        cat EPNr1/procesado.log
  else
        clear
        mensaje="no se encontro el archivo"
fi
        ;;
7)

Procesos=$(ps aux | grep consolidar.sh | grep -v grep)

if [ -n "$PID" ] || [ -n "$Procesos" ] 
then
        if [ -n "$Procesos" ]
           then
             while [ "$respuesta" != "Y" ] && [ "$respuesta" != "N" ]
             do
               echo "Existen procesos antiguos que siguen corriendo, los quiere cerrar?"
               echo " SI(Y) NO(N) "
               read respuesta
               respuesta="${respuesta^^}"
              case $respuesta in
               Y)
                  pkill -f consolidar.sh
                  clear 
                  ;; 
               N)
                  clear
                  ;;
               *)
                clear
                  echo "No ingreso una opcion correcta"
                  ;;
               esac
              done     
        fi  

fi
        clear
        ;;        
*)
clear
        echo "eligio mal"
        ;;
esac
done
  else
    echo "No añadio una variable de entorno para FILENAME"
fi
